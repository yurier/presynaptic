	\label{eq:calcium}
	\begin{align*}
		&\frac{{\rm d}Ca_{pre}}{{\rm d}t} = - \frac{Ca_{pre}}{\tau_{pre}}\\
		&\frac{{\rm d}Ca_{jump}}{{\rm d}t}  = \frac{1-Ca_{jump}}{\tau_{rec}} - \delta_{Ca} \cdot Ca_{jump} \cdot Ca_{pre}\\
		&Ca_{pre} \leftarrow Ca_{pre} + Ca_{jump} \text{ upon spike arrival}
	\end{align*}
